// Install as <project>/ddrp_tests/probe.shader only AFTER starting the
// runtime for the first-library-load test. These programs are unique to it.
include = [ "core/shaders/common.shader" ]
render_states = {
    ddrp_test_state = {
        inherit = "opacity"
        states = { blend_enable = false }
    }
}
bgfx_shaders = {
    ddrp_test_pattern = {
        includes = [ "common" ]
        varying = """
            vec2 v_texcoord0 : TEXCOORD0 = vec2(0.0, 0.0);
            vec3 a_position : POSITION;
            vec2 a_texcoord0 : TEXCOORD0;
        """
        vs_input_output = """
            $input a_position, a_texcoord0
            $output v_texcoord0
        """
        vs_code = """
            void main()
            {
                gl_Position = mul(u_viewProj, vec4(a_position.xy, 0.0, 1.0));
                v_texcoord0 = a_texcoord0;
            }
        """
        fs_input_output = """
            $input v_texcoord0
        """
        fs_code = """
            uniform vec4 u_ddrp_test_color;
            uniform vec4 u_ddrp_pixel_size;
            void main()
            {
                vec2 cell = floor(v_texcoord0 / (8.0 * u_ddrp_pixel_size.xy));
                float checker = mod(cell.x + cell.y, 2.0);
                gl_FragColor = vec4(u_ddrp_test_color.rgb * mix(0.2, 1.0, checker), 1.0);
            }
        """
    }
    ddrp_test_copy = {
        includes = [ "common" ]
        varying = """
            vec2 v_texcoord0 : TEXCOORD0 = vec2(0.0, 0.0);
            vec3 a_position : POSITION;
            vec2 a_texcoord0 : TEXCOORD0;
        """
        vs_input_output = """
            $input a_position, a_texcoord0
            $output v_texcoord0
        """
        vs_code = """
            void main()
            {
                gl_Position = mul(u_viewProj, vec4(a_position.xy, 0.0, 1.0));
                v_texcoord0 = a_texcoord0;
            }
        """
        fs_input_output = """
            $input v_texcoord0
        """
        fs_code = """
            SAMPLER2D(s_color_map, 0);
            void main()
            {
                gl_FragColor = texture2D(s_color_map, v_texcoord0);
            }
        """
    }
}
shaders = {
    ddrp_test_pattern = { bgfx_shader = "ddrp_test_pattern" render_state = "ddrp_test_state" }
    ddrp_test_copy = { bgfx_shader = "ddrp_test_copy" render_state = "ddrp_test_state" }
}
static_compile = [
    { shader = "ddrp_test_pattern" defines = [] }
    { shader = "ddrp_test_copy" defines = [] }
]
